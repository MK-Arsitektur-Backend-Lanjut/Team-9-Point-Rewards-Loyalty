# CURL COMMANDS - Copy & Paste Langsung ke Terminal

# 1. TAMBAH 50 POINTS ke USER 2
curl -X POST http://localhost:8000/api/demo/rewards/add ^
  -H "Content-Type: application/json" ^
  -d "{\"user_id\": 2, \"points\": 50, \"description\": \"Bonus points\"}"

# 2. REDEEM 20 POINTS dari USER 2
curl -X POST http://localhost:8000/api/demo/rewards/redeem ^
  -H "Content-Type: application/json" ^
  -d "{\"user_id\": 2, \"points\": 20, \"description\": \"Redeem for discount\"}"

# 3. CHECK BALANCE USER 2
curl -X GET http://localhost:8000/api/demo/rewards/balance/2

# 4. LIHAT DETAIL BALANCE USER 2
curl -X GET http://localhost:8000/api/demo/rewards/details/2

# 5. VALIDATE BALANCE (check apakah user 2 punya 40 points)
curl -X GET http://localhost:8000/api/demo/rewards/validate/2/40
